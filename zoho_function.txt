void automation.sync_deal_to_operations_and_finance(Int dealId)

	// 1. Fetch Deal / Booking Record
	dealMap = zoho.crm.getRecordById("Deals",dealId.toLong());
	if(dealMap == null || dealMap.isEmpty())
	{
		dealMap = zoho.crm.getRecordById("Potentials",dealId.toLong());
	}

		customerName = ifnull(dealMap.get("Deal_Name"),"");

		endDate = dealMap.get("End_Date");

		}

		// 2. VEHICLE & LOWEST VENDOR RESOLUTION

		vendorLookupId = null;
		totalTransportCost = 0.0;

		{
			vehicleList = zoho.crm.searchRecords("Vehicles","(State:equals:" + destination + ")");

			if(vehicleList != null && vehicleList.size() > 0)
			{
				for each  veh in vehicleList
				{
					minCap = ifnull(veh.get("Min_Capacity"),0).toLong();
					maxCap = ifnull(veh.get("Max_Capacity"),99).toLong();

					{
						matchedVehicle = veh;
						break;
					}
				}
			}
			if(matchedVehicle != null)
			{
				vehicleLookupId = matchedVehicle.get("id");

				lowestCostPerDay = 9999999.0;
				selectedVendorName = "";

				{
					for each  vp in vendorPricingSubform
					{
						cpd = ifnull(vp.get("Cost_Per_day"),0.0).toDecimal();
						if(cpd > 0 && cpd < lowestCostPerDay)
						{
							lowestCostPerDay = cpd;
							selectedVendorName = vp.get("Vendor_Name");
						}
					}
				}
				if(lowestCostPerDay < 9999999.0)
				{
					totalTransportCost = lowestCostPerDay * totalDays;

						vendorSearch = zoho.crm.searchRecords("Vendors","(Vendor_Name:equals:" + selectedVendorName + ")");
						if(vendorSearch != null && vendorSearch.size() > 0)

							vendorLookupId = vendorSearch.get(0).get("id");
						}
					}

			}
		}

			{

				if(ratesList1 != null && ratesList1.size() > 0)
				{
					targetCat = ifnull(dealMap.get("Hotel_1_Category"),roomCategory).toLowerCase();
					for each  row in ratesList1
					{
						rowCat = ifnull(row.get("Room_Category"),"").toLowerCase();
						rPrice = ifnull(row.get("Price_Per_Night"),ifnull(row.get("Base_Price"),ifnull(row.get("Double_Occupancy"),ifnull(row.get("Price"),ifnull(row.get("Rate"),ifnull(row.get("Cost"),0.0)))))).toDecimal();
						if((rowCat == targetCat || rowCat.contains(targetCat) || targetCat.contains(rowCat)) && rPrice > 0)
						{
							pNight1 = rPrice;
							break;
						}
					}
					if(pNight1 == 0.0)
					{
						for each  row in ratesList1
						{
							rPrice = ifnull(row.get("Price_Per_Night"),ifnull(row.get("Base_Price"),ifnull(row.get("Double_Occupancy"),ifnull(row.get("Price"),ifnull(row.get("Rate"),ifnull(row.get("Cost"),0.0)))))).toDecimal();
							if(rPrice > 0)
							{
								pNight1 = rPrice;
								break;
							}
						}
					}
				}
				if(pNight1 == 0.0)
				{
					pNight1 = ifnull(hRec1.get("Price_Per_Night"),ifnull(hRec1.get("Base_Price"),ifnull(hRec1.get("Cost"),0.0))).toDecimal();
				}
				h1Cost = pNight1 * roomsCount * h1Nights;
			}
		}

		h4Cost = 0.0;

			if(hRec4 != null)
			{
				ratesList4 = hRec4.get("Room_Rates_Pricing");
				pNight4 = 0.0;

				{
					targetCat = ifnull(dealMap.get("Hotel_4_Category"),roomCategory).toLowerCase();
					for each  row in ratesList4
					{
						rowCat = ifnull(row.get("Room_Category"),"").toLowerCase();
						rPrice = ifnull(row.get("Price_Per_Night"),ifnull(row.get("Base_Price"),ifnull(row.get("Double_Occupancy"),ifnull(row.get("Price"),ifnull(row.get("Rate"),ifnull(row.get("Cost"),0.0)))))).toDecimal();

						{
							pNight4 = rPrice;
							break;
						}
					}
					if(pNight4 == 0.0)
					{
						for each  row in ratesList4
						{
							rPrice = ifnull(row.get("Price_Per_Night"),ifnull(row.get("Base_Price"),ifnull(row.get("Double_Occupancy"),ifnull(row.get("Price"),ifnull(row.get("Rate"),ifnull(row.get("Cost"),0.0)))))).toDecimal();
							if(rPrice > 0)
							{
								pNight4 = rPrice;
								break;
							}
						}
					}
				}
				if(pNight4 == 0.0)
				{
					pNight4 = ifnull(hRec4.get("Price_Per_Night"),ifnull(hRec4.get("Base_Price"),ifnull(hRec4.get("Cost"),0.0))).toDecimal();

		}
		
		// Hotel 5
		h5Lookup = dealMap.get("th_Hotel1");
		h5Nights = ifnull(dealMap.get("th_Hotel_Nights1"),1).toLong();
		h5Cost = 0.0;
		if(h5Lookup != null && h5Lookup.get("id") != null)

			if(hRec5 != null)
			{
				ratesList5 = hRec5.get("Room_Rates_Pricing");
				pNight5 = 0.0;
				if(ratesList5 != null && ratesList5.size() > 0)

					targetCat = ifnull(dealMap.get("Hotel_5_Category"),roomCategory).toLowerCase();

					{
						rowCat = ifnull(row.get("Room_Category"),"").toLowerCase();
						rPrice = ifnull(row.get("Price_Per_Night"),ifnull(row.get("Base_Price"),ifnull(row.get("Double_Occupancy"),ifnull(row.get("Price"),ifnull(row.get("Rate"),ifnull(row.get("Cost"),0.0)))))).toDecimal();
						if((rowCat == targetCat || rowCat.contains(targetCat) || targetCat.contains(rowCat)) && rPrice > 0)

					}

						{

				}
				if(pNight5 == 0.0)
				{
					pNight5 = ifnull(hRec5.get("Price_Per_Night"),ifnull(hRec5.get("Base_Price"),ifnull(hRec5.get("Cost"),0.0))).toDecimal();

		// Hotel 6
		h6Lookup = dealMap.get("th_Hotel");
		h6Nights = ifnull(dealMap.get("th_Hotel_Nights"),1).toLong();
		h6Cost = 0.0;
		if(h6Lookup != null && h6Lookup.get("id") != null)
		{
			hRec6 = zoho.crm.getRecordById("Hotel",h6Lookup.get("id").toLong());
			if(hRec6 != null)
			{
				ratesList6 = hRec6.get("Room_Rates_Pricing");
				pNight6 = 0.0;
				if(ratesList6 != null && ratesList6.size() > 0)
				{
					targetCat = ifnull(dealMap.get("Hotel_6_Category"),roomCategory).toLowerCase();
					for each  row in ratesList6
					{
						rowCat = ifnull(row.get("Room_Category"),"").toLowerCase();
						rPrice = ifnull(row.get("Price_Per_Night"),ifnull(row.get("Base_Price"),ifnull(row.get("Double_Occupancy"),ifnull(row.get("Price"),ifnull(row.get("Rate"),ifnull(row.get("Cost"),0.0)))))).toDecimal();

						{
							rPrice = ifnull(row.get("Price_Per_Night"),ifnull(row.get("Base_Price"),ifnull(row.get("Double_Occupancy"),ifnull(row.get("Price"),ifnull(row.get("Rate"),ifnull(row.get("Cost"),0.0)))))).toDecimal();

				}
				h6Cost = pNight6 * roomsCount * h6Nights;
			}
		}
		
		// --------------------------------------------------------------------------
		// 4. CREATE / UPDATE OPERATIONS MODULE RECORD
		// --------------------------------------------------------------------------
		opsMap = Map();
		opsMap.put("Name",customerName + " - " + destination + " Tour");
		opsMap.put("Booking_Id",dealIdStr);
		opsMap.put("Customer_Name",customerName);
		opsMap.put("Destination",destination);
		opsMap.put("No_of_Days",totalDays);
		opsMap.put("No_of_Nights",totalNights);
		opsMap.put("No_of_Guests",totalPax);
		opsMap.put("No_of_Adults",adults);
		opsMap.put("Kids_Age",kids);
		opsMap.put("Rooms_Count",roomsCount);
		opsMap.put("Sharing_Type",sharingType);
		opsMap.put("Room_Category",roomCategory);
		if(hotelCategory != "")
		{
			opsMap.put("Hotel_Category",hotelCategory);
		}
		opsMap.put("Trip_Type",tripType);
		opsMap.put("Status","Pending");
		opsMap.put("Itinerary_URL",itineraryUrl);
		if(customerEmail != "")
		{
			opsMap.put("Email",customerEmail);
		}
		
		// Checkboxes
		opsMap.put("Transport_Booked",false);
		opsMap.put("All_Hotels_Booked",false);
		opsMap.put("Activities_Booked",false);
		opsMap.put("Sync_to_Finance",false);
		
		// Itinerary Lookup if present in Deal
		if(dealMap.get("Itinerary") != null && dealMap.get("Itinerary").get("id") != null)
		{
			opsMap.put("Itinerary",dealMap.get("Itinerary").get("id"));
		}
		
		if(startDate != null)
		{
			opsMap.put("Start_Date",startDate);
		}
		if(endDate != null)
		{
			opsMap.put("End_Date",endDate);
		}
		if(vehicleLookupId != null)
		{
			opsMap.put("Vehicle_Type",vehicleLookupId);
		}
		if(vendorLookupId != null)
		{
			opsMap.put("Vendor_Name",vendorLookupId);
		}

		// Hotel 1 in Operations

		opsRes = Map();
		if(existingOps != null && existingOps.size() > 0)
		{
			opsId = existingOps.get(0).get("id");
			opsRes = zoho.crm.updateRecord("Operations",opsId.toLong(),opsMap);
		}
		else
		{
			opsRes = zoho.crm.createRecord("Operations",opsMap);
		}
		info "OPERATIONS_RESPONSE: " + opsRes;
		info "Operations synced successfully for Deal: " + dealIdStr;
	}
