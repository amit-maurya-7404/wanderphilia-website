void automation.sync_deal_to_operations_and_finance(int dealId)
{
	dealIdStr = dealId.toString();
	
	// 1. Fetch Deal / Booking Record
	dealMap = zoho.crm.getRecordById("Deals",dealId.toLong());
	if(dealMap == null || dealMap.isEmpty())
	{
		dealMap = zoho.crm.getRecordById("Potentials",dealId.toLong());
	}
	if(dealMap != null && !dealMap.isEmpty())
	{
		dealName = ifnull(dealMap.get("Deal_Name"),"");
		destination = ifnull(dealMap.get("Destination"),ifnull(dealMap.get("Destinations"),""));
		startDate = dealMap.get("Start_Date");
		endDate = dealMap.get("End_Date");
		
		// Days & Nights
		totalDays = ifnull(dealMap.get("No_of_Days"),0).toLong();
		totalNights = ifnull(dealMap.get("No_of_Nights"),0).toLong();
		if(totalDays <= 0 && startDate != null && endDate != null)
		{
			totalDays = daysBetween(startDate.toDate(),endDate.toDate()) + 1;
			totalNights = if(totalDays > 1,totalDays - 1,1);
		}
		
		// Pax & Room setup
		adults = ifnull(dealMap.get("Number_of_Adults"),ifnull(dealMap.get("Adults"),2)).toLong();
		kids = ifnull(dealMap.get("No_of_Kids"),ifnull(dealMap.get("Children"),0)).toLong();
		totalPax = ifnull(dealMap.get("Number_of_Guests"),ifnull(dealMap.get("No_of_Pax"),adults + kids)).toLong();
		roomsCount = ifnull(dealMap.get("Rooms_Count"),1).toLong();
		if(roomsCount <= 0)
		{
			roomsCount = 1;
		}
		sharingType = ifnull(dealMap.get("Sharing_Type"),"Double Sharing");
		roomCategory = ifnull(dealMap.get("Preferred_Room_Category"),"Deluxe Room");
		hotelCategory = ifnull(dealMap.get("Hotel_Category"),ifnull(dealMap.get("Deal_Hotel_Category"),""));
		tripType = ifnull(dealMap.get("Trip_Type"),"Customised Trip");
		itineraryUrl = ifnull(dealMap.get("Itinerary_URL"),"");
		customerEmail = ifnull(dealMap.get("Email"),ifnull(dealMap.get("Secondary_Email"),""));
		
		// Customer Name as String
		custName = "";
		contactObj = dealMap.get("Contact_Name");
		if(contactObj != null)
		{
			custName = ifnull(contactObj.get("name"),contactObj.toString()).trim();
		}
		if(custName == "")
		{
			custName = dealName;
		}

		// Financials from Deal
		totalPackageCost = ifnull(dealMap.get("Cost"),ifnull(dealMap.get("Amount"),0.0)).toDecimal();
		advancePaid = ifnull(dealMap.get("Adv_Amount"),ifnull(dealMap.get("Advance_Paid"),0.0)).toDecimal();
		balanceDue = ifnull(dealMap.get("Pending_Balance_Amount"),totalPackageCost - advancePaid).toDecimal();
		payStatus = ifnull(dealMap.get("Payment_Status"),if(balanceDue <= 0 && totalPackageCost > 0,"Full Paid",if(advancePaid > 0,"Advance Paid","Pending")));

		// --------------------------------------------------------------------------
		// 2. VEHICLE & LOWEST VENDOR RESOLUTION
		// --------------------------------------------------------------------------
		vehicleLookupId = null;
		vendorLookupId = null;
		selectedVehicleName = ifnull(dealMap.get("Vehicle_Type"),"");
		totalTransportCost = ifnull(dealMap.get("Vehicle_Total_Cost"),ifnull(dealMap.get("Vehicle_Cost"),0.0)).toDecimal();
		if(destination != "" && totalTransportCost == 0.0)
		{
			vehicleList = zoho.crm.searchRecords("Vehicles","(State:equals:" + destination + ")");
			matchedVehicle = null;
			if(vehicleList != null && vehicleList.size() > 0)
			{
				for each  veh in vehicleList
				{
					minCap = ifnull(veh.get("Min_Capacity"),0).toLong();
					maxCap = ifnull(veh.get("Max_Capacity"),99).toLong();
					if(totalPax >= minCap && totalPax <= maxCap)
					{
						matchedVehicle = veh;
						break;
					}
				}
			}
			if(matchedVehicle != null)
			{
				vehicleLookupId = matchedVehicle.get("id");
				selectedVehicleName = ifnull(matchedVehicle.get("Name"),selectedVehicleName);
				vRec = zoho.crm.getRecordById("Vehicles",vehicleLookupId.toLong());
				vendorPricingSubform = vRec.get("Vendor_Pricing");
				lowestCostPerDay = 9999999.0;
				selectedVendorName = "";
				if(vendorPricingSubform != null && vendorPricingSubform.size() > 0)
				{
					for each  vp in vendorPricingSubform
					{
						cpd = ifnull(vp.get("Cost_Per_day"),0.0).toDecimal();
						if(cpd > 0 && cpd < lowestCostPerDay)
						{
							lowestCostPerDay = cpd;
							selectedVendorName = vp.get("Vendor_Name");
						}
					}
				}
				if(lowestCostPerDay < 9999999.0)
				{
					totalTransportCost = lowestCostPerDay * totalDays;
					if(selectedVendorName != "")
					{
						vendorSearch = zoho.crm.searchRecords("Vendors","(Vendor_Name:equals:" + selectedVendorName + ")");
						if(vendorSearch != null && vendorSearch.size() > 0)
						{
							vendorLookupId = vendorSearch.get(0).get("id");
						}
					}
				}
			}
		}

		// --------------------------------------------------------------------------
		// 3. HOTEL 1 TO 6 LOOKUPS, ROOM NAMES & DYNAMIC B2B COST CALCULATION
		// --------------------------------------------------------------------------
		hotelIds = Map();
		hotelNames = Map();
		hotelCities = Map();
		hotelNights = Map();
		hotelDates = Map();
		hotelCategories = Map();
		hotelRoomNames = Map();
		hotelCosts = Map();

		hConfigs = list();
		hConfigs.add({1, "st_Hotel", "st_Hotel_City", "st_Hotel_Nights", "st_Hotel_Stay_Dates", "Hotel_1_Category", "Hotel_1_Room_Name"});
		hConfigs.add({2, "nd_Hotel", "nd_Hotel_City", "nd_Hotel_Nights", "nd_Hotel_Stay_Dates", "Hotel_2_Category", "Hotel_2_Room_Name"});
		hConfigs.add({3, "rd_Hotel", "rd_Hotel_City", "rd_Hotel_Nights", "rd_Hotel_Stay_Dates", "Hotel_3_Category", "Hotel_3_Room_Name"});
		hConfigs.add({4, "th_Hotel2", "th_Hotel_City", "th_Hotel_Nights2", "th_Hotel_Stay_Dates2", "Hotel_4_Category", "Hotel_4_Room_Name"});
		hConfigs.add({5, "th_Hotel1", "th_Hotel_City1", "th_Hotel_Nights1", "th_Hotel_Stay_Dates1", "Hotel_5_Category", "Hotel_5_Room_Name"});
		hConfigs.add({6, "th_Hotel", "th_Hotel_City2", "th_Hotel_Nights", "th_Hotel_Stay_Dates", "Hotel_6_Category", "Hotel_6_Room_Name"});

		for each  cfg in hConfigs
		{
			idx = cfg.get(0);
			hField = cfg.get(1);
			cityField = cfg.get(2);
			nightsField = cfg.get(3);
			datesField = cfg.get(4);
			catField = cfg.get(5);
			roomField = cfg.get(6);

			hObj = ifnull(dealMap.get(hField),dealMap.get("Hotel_" + idx));
			if(hObj != null && hObj.get("id") != null)
			{
				hId = hObj.get("id").toLong();
				hotelIds.put(idx,hId);
				hotelNames.put(idx,ifnull(hObj.get("name"),"Hotel " + idx));
				hotelCities.put(idx,ifnull(dealMap.get(cityField),dealMap.get("Hotel_" + idx + "_City")));
				
				hn = ifnull(dealMap.get(nightsField),ifnull(dealMap.get("Hotel_" + idx + "_Nights"),1)).toLong();
				if(hn <= 0)
				{
					hn = 1;
				}
				hotelNights.put(idx,hn);
				hotelDates.put(idx,ifnull(dealMap.get(datesField),dealMap.get("Hotel_" + idx + "_Dates")));
				hotelCategories.put(idx,ifnull(dealMap.get(catField),hotelCategory));

				rName = ifnull(dealMap.get(roomField),"").toString().trim();
				calculatedCost = 0.0;
				pricePerNight = 0.0;

				hRec = zoho.crm.getRecordById("Hotel",hId);
				if(hRec != null)
				{
					pricingList = hRec.get("Room_Rates_Pricing");
					if(pricingList != null && pricingList.size() > 0)
					{
						if(rName != "" && rName != "null" && rName != "-None-")
						{
							for each  pr in pricingList
							{
								pCat = ifnull(pr.get("Hotel_Room_Category"),"").toString().trim();
								if(pCat.equalsIgnoreCase(rName))
								{
									pricePerNight = ifnull(pr.get("Price_Per_Night"),ifnull(pr.get("Base_Price"),0.0)).toDecimal();
									break;
								}
							}
						}
						if(pricePerNight == 0.0)
						{
							firstPr = pricingList.get(0);
							if(rName == "" || rName == "null" || rName == "-None-")
							{
								rName = ifnull(firstPr.get("Hotel_Room_Category"),"Deluxe Room").toString().trim();
							}
							pricePerNight = ifnull(firstPr.get("Price_Per_Night"),ifnull(firstPr.get("Base_Price"),0.0)).toDecimal();
						}
					}
				}
				if(rName == "" || rName == "null")
				{
					rName = "Deluxe Room";
				}
				hotelRoomNames.put(idx,rName);

				calculatedCost = pricePerNight * roomsCount * hn;
				hotelCosts.put(idx,calculatedCost);
			}
		}

		// --------------------------------------------------------------------------
		// 4. MAP DATA TO OPERATIONS
		// --------------------------------------------------------------------------
		opsMap = Map();
		opsMap.put("Booking_Id",dealIdStr);
		opsMap.put("Name",dealName);
		if(custName != "")
		{
			opsMap.put("Customer_Name",custName);
		}
		opsMap.put("Destination",destination);
		opsMap.put("Start_Date",startDate);
		opsMap.put("End_Date",endDate);
		opsMap.put("No_of_Days",totalDays);
		opsMap.put("No_of_Nights",totalNights);
		opsMap.put("No_of_Guests",totalPax);
		opsMap.put("No_of_Adults",adults);
		opsMap.put("Kids_Age",kids);
		if(dealMap.get("Child_1_Age") != null) { opsMap.put("Child_1_Age",dealMap.get("Child_1_Age")); }
		if(dealMap.get("Child_1_Bed_Preference") != null) { opsMap.put("Child_1_Bed_Preference",dealMap.get("Child_1_Bed_Preference")); }
		opsMap.put("Rooms_Count",roomsCount);
		opsMap.put("Sharing_Type",sharingType);
		opsMap.put("Room_Category",roomCategory);
		if(hotelCategory != "")
		{
			opsMap.put("Hotel_Category",hotelCategory);
		}
		opsMap.put("Trip_Type",tripType);
		opsMap.put("Status","Pending");
		opsMap.put("Itinerary_URL",itineraryUrl);
		if(customerEmail != "")
		{
			opsMap.put("Email",customerEmail);
		}
		if(dealMap.get("Secondary_Email") != null)
		{
			opsMap.put("Secondary_Email",dealMap.get("Secondary_Email"));
		}
		if(dealMap.get("Itinerary") != null && dealMap.get("Itinerary").get("id") != null)
		{
			opsMap.put("Itinerary",dealMap.get("Itinerary").get("id"));
		}

		// Transport
		if(dealMap.get("Vehicle_Type") != null)
		{
			opsMap.put("Vehicle_Type",dealMap.get("Vehicle_Type"));
		}
		else if(vehicleLookupId != null)
		{
			opsMap.put("Vehicle_Type",vehicleLookupId);
		}
		if(vendorLookupId != null)
		{
			opsMap.put("Vendor_Name",vendorLookupId);
		}
		if(totalTransportCost > 0)
		{
			opsMap.put("Vehicle_Cost",totalTransportCost);
		}

		// Hotels 1 to 6
		for each  idx in {1,2,3,4,5,6}
		{
			if(hotelIds.containsKey(idx))
			{
				opsMap.put("Hotel_" + idx,hotelIds.get(idx));
				opsMap.put("Hotel_" + idx + "_City",hotelCities.get(idx));
				opsMap.put("Hotel_" + idx + "_Category",hotelCategories.get(idx));
				opsMap.put("Hotel_" + idx + "_Nights",hotelNights.get(idx));
				opsMap.put("Hotel_" + idx + "_Dates",hotelDates.get(idx));
				opsMap.put("Hotel_" + idx + "_Cost",hotelCosts.get(idx));
				if(hotelRoomNames.get(idx) != null && hotelRoomNames.get(idx) != "")
				{
					opsMap.put("Hotel_" + idx + "_Room_Name",hotelRoomNames.get(idx));
				}
			}
		}

		// Activities Subform Sync
		dealActivities = dealMap.get("activities_and_experince_Details");
		if(dealActivities == null || dealActivities.isEmpty())
		{
			dealActivities = dealMap.get("Activities_and_Experiences_operations");
		}
		if(dealActivities == null || dealActivities.isEmpty())
		{
			dealActivities = dealMap.get("Activities_and_Experiences");
		}
		if(dealActivities != null && dealActivities.size() > 0)
		{
			actRows = list();
			for each  act in dealActivities
			{
				rowMap = Map();
				if(act.get("Days") != null) { rowMap.put("Days",act.get("Days")); }
				if(act.get("City") != null) { rowMap.put("City",act.get("City")); }
				if(act.get("City_2") != null) { rowMap.put("City_2",act.get("City_2")); }
				if(act.get("Guided_Tour") != null) { rowMap.put("Guided_Tour",act.get("Guided_Tour")); }
				enRoute = ifnull(act.get("En_Route_Experinces"),act.get("En_route_Experiences"));
				if(enRoute != null)
				{
					rowMap.put("En_Route_Experinces",ifnull(enRoute.get("id"),enRoute));
				}
				for each  expIdx in {1,2,3,4}
				{
					expVal = act.get("Experience_" + expIdx);
					if(expVal != null)
					{
						rowMap.put("Experience_" + expIdx,ifnull(expVal.get("id"),expVal));
					}
				}
				actRows.add(rowMap);
			}
			if(actRows.size() > 0)
			{
				opsMap.put("Activities_and_Experiences_operations",actRows);
			}
		}

		// Save Operations
		existingOps = zoho.crm.searchRecords("Operations","(Booking_Id:equals:" + dealIdStr + ")");
		if(existingOps != null && existingOps.size() > 0)
		{
			opsId = existingOps.get(0).get("id");
			opsRes = zoho.crm.updateRecord("Operations",opsId.toLong(),opsMap);
			info "Operations Updated: " + opsRes;
		}
		else
		{
			opsRes = zoho.crm.createRecord("Operations",opsMap);
			info "Operations Created: " + opsRes;
		}

		// --------------------------------------------------------------------------
		// 5. MAP CLIENT PAYMENT & DETAILS TO FINANCE
		// --------------------------------------------------------------------------
		finMap = Map();
		finMap.put("Booking_ID",dealIdStr);
		finMap.put("Name",dealName);
		finMap.put("Total_Receivable_Amount",totalPackageCost);
		finMap.put("Total_Amount_Received",advancePaid);
		finMap.put("Balance_Receivable",balanceDue);
		finMap.put("Payment_Status",payStatus);
		finMap.put("Status","Pending");
		
		if(vehicleLookupId != null)
		{
			finMap.put("Vehicle_Name",vehicleLookupId);
		}
		if(vendorLookupId != null)
		{
			finMap.put("Vendor_Name",vendorLookupId);
		}
		if(totalTransportCost > 0)
		{
			finMap.put("Total_Transport_Cost",totalTransportCost);
		}

		if(hotelIds.containsKey(1)) { finMap.put("Hotel_Name",hotelIds.get(1)); finMap.put("Total_Hotel_Cost",hotelCosts.get(1)); }
		if(hotelIds.containsKey(2)) { finMap.put("Hotel_2_Name",hotelIds.get(2)); finMap.put("Total_Hotel_2_Cost",hotelCosts.get(2)); }
		if(hotelIds.containsKey(3)) { finMap.put("Hotel_3_Name",hotelIds.get(3)); finMap.put("Total_Hotel_3_Cost",hotelCosts.get(3)); }
		if(hotelIds.containsKey(4)) { finMap.put("Hotel_4_Name",hotelIds.get(4)); finMap.put("Total_Hotel_4_Cost",hotelCosts.get(4)); }
		if(hotelIds.containsKey(5)) { finMap.put("Hotel_5_Name",hotelIds.get(5)); finMap.put("Total_Hotel_5_Cost",hotelCosts.get(5)); }
		if(hotelIds.containsKey(6)) { finMap.put("Hotel_6_Name",hotelIds.get(6)); finMap.put("Total_Hotel_6_Cost",hotelCosts.get(6)); }

		// Save Finance
		existingFin = zoho.crm.searchRecords("Finance","(Booking_ID:equals:" + dealIdStr + ")");
		if(existingFin != null && existingFin.size() > 0)
		{
			finId = existingFin.get(0).get("id");
			finRes = zoho.crm.updateRecord("Finance",finId.toLong(),finMap);
			info "Finance Updated: " + finRes;
		}
		else
		{
			finRes = zoho.crm.createRecord("Finance",finMap);
			info "Finance Created: " + finRes;
		}

		// --------------------------------------------------------------------------
		// 6. SEND EMAIL ALERTS TO OPERATIONS & FINANCE
		// --------------------------------------------------------------------------
		try 
		{
			// Operations Email Body
			opsEmailContent = "<html><body style='font-family:Arial,sans-serif;line-height:1.6;color:#333;'>";
			opsEmailContent = opsEmailContent + "<h2 style='color:#0d6efd;'>New Booking Alert - Operations Team</h2>";
			opsEmailContent = opsEmailContent + "<p>A new booking has been confirmed. Below are the trip and operations details:</p>";
			opsEmailContent = opsEmailContent + "<table border='1' cellpadding='8' cellspacing='0' style='border-collapse:collapse;width:100%;max-width:600px;'>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Booking ID</b></td><td>" + dealIdStr + "</td></tr>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Deal Name</b></td><td>" + dealName + "</td></tr>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Customer Name</b></td><td>" + custName + "</td></tr>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Destination</b></td><td>" + destination + "</td></tr>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Travel Dates</b></td><td>" + startDate + " to " + endDate + " (" + totalDays + " Days / " + totalNights + " Nights)</td></tr>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Total Guests</b></td><td>" + totalPax + " (" + adults + " Adults, " + kids + " Kids)</td></tr>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Rooms & Sharing</b></td><td>" + roomsCount + " Room(s), " + sharingType + " (" + roomCategory + ")</td></tr>";
			opsEmailContent = opsEmailContent + "<tr><td style='background:#f4f4f4;'><b>Vehicle Type</b></td><td>" + selectedVehicleName + "</td></tr>";
			opsEmailContent = opsEmailContent + "</table>";

			// Hotel breakdown in Operations Email
			if(hotelIds.size() > 0)
			{
				opsEmailContent = opsEmailContent + "<h3 style='margin-top:20px;'>Hotel Reservations Required:</h3>";
				opsEmailContent = opsEmailContent + "<table border='1' cellpadding='8' cellspacing='0' style='border-collapse:collapse;width:100%;max-width:600px;'>";
				opsEmailContent = opsEmailContent + "<tr style='background:#f4f4f4;'><th>#</th><th>Hotel Name</th><th>City</th><th>Room Category</th><th>Stay Dates</th></tr>";
				for each  idx in {1,2,3,4,5,6}
				{
					if(hotelIds.containsKey(idx))
					{
						opsEmailContent = opsEmailContent + "<tr><td>" + idx + "</td><td>" + hotelNames.get(idx) + "</td><td>" + hotelCities.get(idx) + "</td><td>" + hotelRoomNames.get(idx) + "</td><td>" + hotelDates.get(idx) + "</td></tr>";
					}
				}
				opsEmailContent = opsEmailContent + "</table>";
			}
			opsEmailContent = opsEmailContent + "<p style='margin-top:20px;'>Please initiate supplier bookings accordingly.</p>";
			opsEmailContent = opsEmailContent + "</body></html>";

			sendmail
			[
				from: zoho.loginuserid
				to: "operations@wanderphilia.com"
				subject: "New Booking Confirmed: " + dealName + " (" + destination + ")"
				message: opsEmailContent
			]
			info "Operations email sent successfully.";

			// Finance Email Body
			finEmailContent = "<html><body style='font-family:Arial,sans-serif;line-height:1.6;color:#333;'>";
			finEmailContent = finEmailContent + "<h2 style='color:#198754;'>New Booking Payment & Cost Summary - Finance Team</h2>";
			finEmailContent = finEmailContent + "<p>A new booking has been confirmed. Below are the client payment and vendor costing details:</p>";
			finEmailContent = finEmailContent + "<table border='1' cellpadding='8' cellspacing='0' style='border-collapse:collapse;width:100%;max-width:600px;'>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Booking ID</b></td><td>" + dealIdStr + "</td></tr>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Customer Name</b></td><td>" + custName + "</td></tr>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Destination</b></td><td>" + destination + "</td></tr>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Total Package Amount</b></td><td><b>INR " + totalPackageCost + "</b></td></tr>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Advance Received</b></td><td style='color:green;'><b>INR " + advancePaid + "</b></td></tr>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Balance Receivable</b></td><td style='color:red;'><b>INR " + balanceDue + "</b></td></tr>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Payment Status</b></td><td>" + payStatus + "</td></tr>";
			finEmailContent = finEmailContent + "<tr><td style='background:#f4f4f4;'><b>Transport B2B Cost</b></td><td>INR " + totalTransportCost + "</td></tr>";
			finEmailContent = finEmailContent + "</table>";

			if(hotelCosts.size() > 0)
			{
				finEmailContent = finEmailContent + "<h3 style='margin-top:20px;'>Hotel B2B Costs Payable:</h3>";
				finEmailContent = finEmailContent + "<table border='1' cellpadding='8' cellspacing='0' style='border-collapse:collapse;width:100%;max-width:600px;'>";
				finEmailContent = finEmailContent + "<tr style='background:#f4f4f4;'><th>#</th><th>Hotel Name</th><th>Nights</th><th>B2B Cost</th></tr>";
				for each  idx in {1,2,3,4,5,6}
				{
					if(hotelIds.containsKey(idx))
					{
						finEmailContent = finEmailContent + "<tr><td>" + idx + "</td><td>" + hotelNames.get(idx) + "</td><td>" + hotelNights.get(idx) + "</td><td>INR " + hotelCosts.get(idx) + "</td></tr>";
					}
				}
				finEmailContent = finEmailContent + "</table>";
			}
			finEmailContent = finEmailContent + "<p style='margin-top:20px;'>Please verify client receipts and plan vendor payouts.</p>";
			finEmailContent = finEmailContent + "</body></html>";

			sendmail
			[
				from: zoho.loginuserid
